cmake_minimum_required(VERSION 3.16)
project(vn_cubism_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(OpenGL REQUIRED)
find_package(GLEW REQUIRED)

# Make a private build copy. Never modify the user's SDK. The official renderer
# uses client-side arrays; raylib's GL 3.3 core context requires VBOs/VAOs.
file(COPY "${CUBISM_SDK_ROOT}/Framework/" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/sdk-framework")
foreach(name CubismShader_OpenGLES2 CubismRenderer_OpenGLES2)
    set(path "${CMAKE_CURRENT_BINARY_DIR}/sdk-framework/src/Rendering/OpenGL/${name}.cpp")
    file(READ "${path}" source)
    string(ASCII 239 187 191 bom)
    string(REPLACE "${bom}" "" source "${source}")
    string(REPLACE "glVertexAttribPointer(" "vn_vertex_attrib_pointer(" source "${source}")
    string(REPLACE "glDrawElements(" "vn_draw_elements(" source "${source}")
    file(WRITE "${path}" "#include \"core_profile.hpp\"\n${source}")
endforeach()
set(FRAMEWORK_SOURCE OpenGL)
add_subdirectory("${CMAKE_CURRENT_BINARY_DIR}/sdk-framework" Framework)
target_compile_definitions(Framework PUBLIC CSM_TARGET_LINUX_GL)
target_include_directories(Framework PUBLIC "${CUBISM_SDK_ROOT}/Core/include" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(Framework PUBLIC GLEW::GLEW OpenGL::GL)
add_library(vn_cubism_bridge STATIC bridge.cpp core_profile.cpp)
target_include_directories(vn_cubism_bridge PRIVATE "${VN_GENERATED}")
target_link_libraries(vn_cubism_bridge PRIVATE Framework)
